<!-- Scripts. 1: scroll-to-form fix that clears a fixed site header.
     2: days-to-go label. 3: phone bar. The page still works if
     these are stripped. --><script type="text/javascript">/*<![CDATA[*/
(function(){
  var target = document.getElementById('cmc-rsvp');
  if (!target || !target.getBoundingClientRect) { return; }

  /* 1. Scroll to the form, clearing any fixed or sticky site header */
  function headerOffset(){
    var h = 0, el, w, cs;
    w = window.innerWidth || document.documentElement.clientWidth;
    if (!document.elementFromPoint) { return 0; }
    el = document.elementFromPoint(Math.floor(w / 2), 2);
    while (el && el !== document.body && el !== document.documentElement) {
      if (target.contains(el)) { return 0; }
      cs = window.getComputedStyle ? window.getComputedStyle(el) : null;
      if (cs && (cs.position === 'fixed' || cs.position === 'sticky')) {
        h = el.getBoundingClientRect().bottom;
        break;
      }
      el = el.parentNode;
    }
    return h > 0 ? h : 0;
  }

  function align(){
    var off = headerOffset();
    var delta = target.getBoundingClientRect().top - off;
    if (Math.abs(delta) > 2) { window.scrollBy(0, delta); }
  }

  function go(){
    target.scrollIntoView(true);
    align();
    setTimeout(align, 120);
    setTimeout(align, 450);
    setTimeout(align, 1000);
  }

  var links = document.querySelectorAll ? document.querySelectorAll('.cmc-wrap a[href="#cmc-rsvp"]') : [];
  for (var i = 0; i < links.length; i++) {
    links[i].addEventListener('click', function(e){
      e.preventDefault();
      go();
    }, false);
  }

  if (window.location.hash === '#cmc-rsvp') {
    window.addEventListener('load', function(){ setTimeout(go, 50); }, false);
  }

  /* 2. Days-to-go label. Leaves the plain date in place once the day has passed */
  var count = document.getElementById('cmc-count');
  if (count) {
    var now = new Date();
    var today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
    var eventDay = new Date(2026, 11, 6);
    var days = Math.round((eventDay.getTime() - today.getTime()) / 86400000);
    if (days > 1) { count.innerHTML = days + ' days to go'; }
    else if (days === 1) { count.innerHTML = 'Tomorrow at 5 PM'; }
    else if (days === 0) { count.innerHTML = 'Today at 5 PM'; }
  }

  /* 3. Phone bar: on only while neither the top button nor the form is on screen */
  var bar = document.getElementById('cmc-sticky');
  var heroBtn = document.getElementById('cmc-hero-btn');
  if (bar && heroBtn && ('IntersectionObserver' in window)) {
    var heroIn = true, formIn = false;
    var io = new IntersectionObserver(function(entries){
      for (var j = 0; j < entries.length; j++) {
        if (entries[j].target === heroBtn) { heroIn = entries[j].isIntersecting; }
        if (entries[j].target === target) { formIn = entries[j].isIntersecting; }
      }
      bar.className = (!heroIn && !formIn) ? 'cmc-sticky cmc-on' : 'cmc-sticky';
    }, { threshold: 0 });
    io.observe(heroBtn);
    io.observe(target);
  }
})();
